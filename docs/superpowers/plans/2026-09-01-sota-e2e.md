```c
hu_sqlite_qc_t hu_sqlite_quick_check(sqlite3 *db) {
    if (!db) return HU_QC_INDETERMINATE;
    /* A WAL with a pending checkpoint is not corruption; fold it in first so
     * quick_check sees the committed state. Ignore the rc: busy is fine. */
    sqlite3_wal_checkpoint_v2(db, NULL, SQLITE_CHECKPOINT_PASSIVE, NULL, NULL);
    for (int attempt = 0; attempt < 3; attempt++) {
        sqlite3_stmt *stmt = NULL;
        int prc = sqlite3_prepare_v2(db, "PRAGMA quick_check;", -1, &stmt, NULL);
        if (prc == SQLITE_NOTADB || prc == SQLITE_CORRUPT) return HU_QC_CORRUPT;
        if (prc != SQLITE_OK) { if (prc == SQLITE_BUSY || prc == SQLITE_LOCKED) { hu_platform_sleep_ms(200); continue; } return HU_QC_INDETERMINATE; }
        int src = sqlite3_step(stmt);
        hu_sqlite_qc_t r = HU_QC_INDETERMINATE;
        if (src == SQLITE_ROW) {
            const unsigned char *row = sqlite3_column_text(stmt, 0);
            r = (row && strcmp((const char *)row, "ok") == 0) ? HU_QC_OK : HU_QC_CORRUPT;
        } else if (src == SQLITE_CORRUPT || src == SQLITE_NOTADB) r = HU_QC_CORRUPT;
        sqlite3_finalize(stmt);
        if (r != HU_QC_INDETERMINATE || (src != SQLITE_BUSY && src != SQLITE_LOCKED)) return r;
        hu_platform_sleep_ms(200);
    }
    return HU_QC_INDETERMINATE;
}
bool hu_sqlite_quick_check_ok(sqlite3 *db) { return hu_sqlite_quick_check(db) == HU_QC_OK; }
```

Call site: quarantine only on `HU_QC_CORRUPT`; on `HU_QC_INDETERMINATE` log `hu_log_warn("memory.sqlite", …, "quick_check indeterminate (busy/locked) — NOT quarantining '%s'; will re-check next boot")` and leave the sentinel in place.
